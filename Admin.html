function downloadPDF() { if (!visibleAttendance || visibleAttendance.length === 0) { alert("PDF ke liye koi attendance record nahi mila."); return; } try { const teacher = document.getElementById("teacherSearch").value.trim(); const date = document.getElementById("dateSearch").value.trim(); let content = ""; content += "MOON STAR PUBLIC HIGH SCHOOL KHANWAWAN\n"; content += "TEACHER ATTENDANCE REPORT\n\n"; if (teacher) { content += "Teacher: " + teacher + "\n"; } if (date) { content += "Date: " + date + "\n"; } content += "\n"; content += "Teacher | Status | Date | Time\n"; content += "------------------------------------------\n"; visibleAttendance.slice().reverse().forEach(function(record) { content += record.name + " | " + record.status + " | " + record.date + " | " + record.time + "\n"; }); content += "\nTotal Records: " + visibleAttendance.length; const blob = new Blob([content], { type: "text/plain;charset=utf-8" }); const url = URL.createObjectURL(blob); const a = document.createElement("a"); a.href = url; a.download = "Teacher_Attendance_Report.txt"; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url); } catch (error) { alert("Download error: " + error.message); } }
